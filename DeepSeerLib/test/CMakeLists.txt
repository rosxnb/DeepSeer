include(GoogleTest)

function(deepseer_add_test name source)
    add_executable(${name} ${source})
    target_link_libraries(${name} PRIVATE DeepSeerLib GTest::gtest_main)
    deepseer_set_compiler_options(${name})
    deepseer_enable_sanitizers(${name})
    gtest_discover_tests(${name})
endfunction()

deepseer_add_test(TestBuffer        Core/TestBuffer.cpp)
deepseer_add_test(TestEventLoop     Event/TestEventLoop.cpp)
deepseer_add_test(TestLogger        Log/TestLogger.cpp)
deepseer_add_test(TestHttp1Codec    Http/TestHttp1Codec.cpp)
deepseer_add_test(TestConnection    Net/TestConnection.cpp)
